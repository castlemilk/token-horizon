// Review probe 4: re-run flipped / borderline artifacts with the NEW shipped checks,
// keep the frames each check read, and write crops so a reviewer can judge by eye.
// Usage: tsx t4-flips.mts <outDir> <task|model|check> ...
import { mkdirSync, readFileSync, writeFileSync } from 'node:fs'
import { deflateSync } from 'node:zlib'
import { BENCHMARK_TASKS } from '<benchmarks>/lib/lab/llm-benchmark/registry.ts'
import { getChecksForTask } from '<benchmarks>/lib/lab/llm-benchmark/scorers/checks.ts'
import { closeSandbox, runChecks, type CheckContext, type CheckFn } from '<benchmarks>/lib/lab/llm-benchmark/scorers/sandbox.ts'
import { decodePng, maxChannelDifference, type DecodedImage } from '<benchmarks>/lib/lab/llm-benchmark/scorers/pixels.ts'

const [, , outDir, ...specs] = process.argv
mkdirSync(outDir!, { recursive: true })
const records = JSON.parse(
  readFileSync('<benchmarks>/lib/lab/llm-benchmark/results.json', 'utf8')
) as Array<{ taskId: string; modelId: string; output: string; source: string }>

// minimal RGBA PNG writer (filter 0)
const CRC = (() => {
  const t = new Uint32Array(256)
  for (let n = 0; n < 256; n++) {
    let c = n
    for (let k = 0; k < 8; k++) c = c & 1 ? 0xedb88320 ^ (c >>> 1) : c >>> 1
    t[n] = c >>> 0
  }
  return t
})()
function crc(b: Uint8Array) {
  let c = 0xffffffff
  for (const x of b) c = CRC[(c ^ x) & 0xff]! ^ (c >>> 8)
  return (c ^ 0xffffffff) >>> 0
}
function chunk(type: string, body: Uint8Array) {
  const head = Buffer.alloc(8)
  head.writeUInt32BE(body.length, 0)
  head.write(type, 4, 'latin1')
  const c = Buffer.alloc(4)
  c.writeUInt32BE(crc(Buffer.concat([Buffer.from(type, 'latin1'), Buffer.from(body)])), 0)
  return Buffer.concat([head, Buffer.from(body), c])
}
function writePng(path: string, w: number, h: number, rgba: Uint8Array) {
  const raw = Buffer.alloc(h * (w * 4 + 1))
  for (let y = 0; y < h; y++) Buffer.from(rgba.buffer, rgba.byteOffset + y * w * 4, w * 4).copy(raw, y * (w * 4 + 1) + 1)
  const ihdr = Buffer.alloc(13)
  ihdr.writeUInt32BE(w, 0)
  ihdr.writeUInt32BE(h, 4)
  ihdr[8] = 8
  ihdr[9] = 6
  writeFileSync(
    path,
    Buffer.concat([Buffer.from([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]), chunk('IHDR', ihdr), chunk('IDAT', deflateSync(raw)), chunk('IEND', new Uint8Array(0))])
  )
}

/** before | after | mask, cropped to the changed bbox (+pad), upscaled so it is readable. */
function sideBySide(a: DecodedImage, b: DecodedImage, box: { x0: number; y0: number; x1: number; y1: number }, pad: number, path: string) {
  const x0 = Math.max(0, box.x0 - pad)
  const y0 = Math.max(0, box.y0 - pad)
  const x1 = Math.min(a.width - 1, box.x1 + pad)
  const y1 = Math.min(a.height - 1, box.y1 + pad)
  const cw = x1 - x0 + 1
  const ch = y1 - y0 + 1
  const scale = Math.max(1, Math.min(8, Math.floor(420 / Math.max(cw, ch))))
  const gap = 6
  const W = (cw * scale) * 3 + gap * 2
  const H = ch * scale
  const out = new Uint8Array(W * H * 4).fill(255)
  for (let y = 0; y < ch; y++)
    for (let x = 0; x < cw; x++) {
      const p = (y0 + y) * a.width + (x0 + x)
      const changed = maxChannelDifference(a.data, b.data, p) > 8
      const src = [a.data.subarray(p * 4, p * 4 + 4), b.data.subarray(p * 4, p * 4 + 4), changed ? [255, 0, 0, 255] : [0, 0, 0, 255]]
      for (let k = 0; k < 3; k++)
        for (let sy = 0; sy < scale; sy++)
          for (let sx = 0; sx < scale; sx++) {
            const o = ((y * scale + sy) * W + k * (cw * scale + gap) + x * scale + sx) * 4
            out[o] = src[k]![0]!
            out[o + 1] = src[k]![1]!
            out[o + 2] = src[k]![2]!
            out[o + 3] = 255
          }
    }
  writePng(path, W, H, out)
  return { crop: `${x0},${y0}-${x1},${y1}`, scale }
}

for (const spec of specs) {
  const [taskId, modelId, checkName] = spec.split('|') as [string, string, string]
  const rec = records.find((r) => r.taskId === taskId && r.modelId === modelId && r.source !== 'seeded')
  if (!rec) {
    console.log('MISSING', spec)
    continue
  }
  const task = BENCHMARK_TASKS.find((t) => t.id === taskId)!
  const checks = getChecksForTask(task)
  const idx = checks.findIndex((c) => c.checkName === checkName)
  const caps: Array<Buffer | null> = []
  const slug = `${checkName}__${modelId}`.replace(/[^a-z0-9_.-]/gi, '_')
  const wrapped: CheckFn[] = checks.map((check, i) => {
    const w = async (ctx: CheckContext) => {
      const cap = ctx.captureCanvas
      if (i === idx) await ctx.page.screenshot({ path: `${outDir}/${slug}.page-before.png` }).catch(() => {})
      const res = await check({
        ...ctx,
        captureCanvas: async (sel?: string) => {
          const r = await cap(sel)
          if (i === idx) caps.push(r ? r.data : null)
          return r
        },
      })
      if (i === idx) await ctx.page.screenshot({ path: `${outDir}/${slug}.page-after.png` }).catch(() => {})
      return res
    }
    return Object.assign(w, { checkName: check.checkName, maxPoints: check.maxPoints })
  })
  const results = await runChecks(rec.output, wrapped, { settleMs: 600, perCheckTimeoutMs: 6000, totalTimeoutMs: 25_000 })
  const res = results[idx]
  let line = `${spec} => passed=${res?.passed} ${res?.points}/${res?.maxPoints} :: ${res?.detail ?? ''}`
  caps.forEach((c, k) => c && writeFileSync(`${outDir}/${slug}.c${k}.png`, c))
  if (caps[0] && caps[1]) {
    const a = decodePng(caps[0])
    const b = decodePng(caps[1])
    if (a.width === b.width && a.height === b.height) {
      let x0 = a.width, y0 = a.height, x1 = -1, y1 = -1, n = 0, n0 = 0, n32 = 0
      for (let y = 0; y < a.height; y++)
        for (let x = 0; x < a.width; x++) {
          const d = maxChannelDifference(a.data, b.data, y * a.width + x)
          if (d > 0) n0++
          if (d > 32) n32++
          if (d > 8) {
            n++
            if (x < x0) x0 = x
            if (x > x1) x1 = x
            if (y < y0) y0 = y
            if (y > y1) y1 = y
          }
        }
      line += ` || ${a.width}x${a.height} changed>8=${n}px (${((100 * n) / (a.width * a.height)).toFixed(3)}%) >0=${n0}px >32=${n32}px bbox=${n ? `${x0},${y0}-${x1},${y1}` : 'none'}`
      if (n) {
        const info = sideBySide(a, b, { x0, y0, x1, y1 }, 12, `${outDir}/${slug}.crop.png`)
        line += ` crop=${info.crop} x${info.scale}`
      }
    } else line += ` || size changed ${a.width}x${a.height} -> ${b.width}x${b.height}`
  }
  console.log(line)
}
await closeSandbox()
